package data

import (
 "encoding/json"
 "errors"
 "os"
 "testing"
 biz "MODULE/BIZ"
 "MODULE/DATA/sqlcgen"
 "k8s.io/apimachinery/pkg/apis/meta/v1/unstructured"
)

// This read-only diagnostic consumes a frozen real-object inventory. It does not
// alter the application, cluster, database, or published test collection.
func TestExistingManualPublicPoolReuseDiagnostic(t *testing.T) {
 raw, err := os.ReadFile(os.Getenv("REUSE_INVENTORY")); if err != nil {t.Fatal(err)}
 var inventory map[string]json.RawMessage
 if err=json.Unmarshal(raw,&inventory); err != nil {t.Fatal(err)}
 objects:=map[string]*unstructured.Unstructured{}
 for key,kind:=range map[string]string{"pool":"public_pool","gateway":"egress_gateway","vlan":"vlan"} {
  t.Run(key,func(t *testing.T){
   o:=&unstructured.Unstructured{}
   if err:=o.UnmarshalJSON(inventory[key]);err!=nil {t.Fatal(err)}
   objects[key]=o
   // IDs below only construct a negative inspector input; nothing is registered.
   binding:=sqlcgen.NetworkProviderBinding{ResourceKind:kind,Namespace:o.GetNamespace(),ProviderName:o.GetName(),ProviderUid:string(o.GetUID()),BindingID:"diagnostic-only-no-registration"}
   err:=inspectEgressIdentity(o,binding,"diagnostic-only-resource",string(o.GetUID()))
   var providerError *biz.ProviderError
   if !errors.As(err,&providerError)||providerError.Kind!=biz.ProviderConflict {t.Fatalf("want identity conflict for unregistered manual object, got %v",err)}
   if o.GetLabels()[ownerLabel]!="" || o.GetLabels()[bindingLabel]!="" || o.GetLabels()[resourceLabel]!="" {t.Fatal("manual object already has Network registration; diagnosis must be revised")}
   t.Logf("observed %s %s UID=%s: Network ownership labels absent, existing inspector rejects",kind,o.GetName(),o.GetUID())
  })
 }
 pool:=objects["pool"]
 actual,_,_:=unstructured.NestedString(pool.Object,"spec","allowedNamespaces","from")
 if actual!="Selector" {t.Fatalf("manual selector changed: %s",actual)}
 upstream:="172.16.102.1"
 expected:=renderPublicPool(sqlcgen.NetworkPublicPool{Mode:"underlay",Scope:"public",Cidr:"172.16.102.0/24",OvnGatewayIp:"172.16.102.2",UpstreamGatewayIp:&upstream,ExcludedIps:[]string{"172.16.102.1..172.16.102.191","172.16.102.208..172.16.102.254"}},objects["gateway"].GetName(),objects["vlan"].GetName())
 if matchesEgressSpec(pool,expected,"public_pool") {t.Fatal("unexpectedly accepts existing namespace selector")}
 t.Logf("actual namespace mode=%s, product-rendered=%v; existing pool cannot be passed through unchanged",actual,expected["allowedNamespaces"])
}
